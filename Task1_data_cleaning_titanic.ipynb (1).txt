        "import pandas as pd\n",
        "\n",
        "# ---------- Step 1: Load the data ----------\n",
        "df = pd.read_csv(\"titanic.csv\")\n",
        "print(\"Shape of data:\", df.shape)\n",
        "print(df.head())\n",
        "\n",
        "change_log = []\n",
        "\n",
        "# ---------- Step 2: Check the problems first ----------\n",
        "print(\"\\n--- Info ---\")\n",
        "df.info()\n",
        "print(\"\\n--- Null values ---\")\n",
        "print(df.isnull().sum())\n",
        "print(\"\\n--- Duplicate rows ---\")\n",
        "print(df.duplicated().sum())\n",
        "\n",
        "# ---------- Step 3: Remove duplicates ----------\n",
        "dup_count = df.duplicated().sum()\n",
        "df = df.drop_duplicates()\n",
        "change_log.append([\"Duplicate rows\", \"Removed with drop_duplicates()\", dup_count])\n",
        "\n",
        "# ---------- Step 4: Fix text columns ----------\n",
        "df[\"Name\"] = df[\"Name\"].str.strip()\n",
        "df[\"Sex\"] = df[\"Sex\"].str.strip().str.lower()\n",
        "df[\"Embarked\"] = df[\"Embarked\"].str.strip().str.upper()\n",
        "change_log.append([\"Extra spaces / different casing in text\", \"Used strip(), lower() and upper()\", \"Name, Sex, Embarked\"])\n",
        "\n",
        "# ---------- Step 5: Handle missing values ----------\n",
        "age_nulls = df[\"Age\"].isnull().sum()\n",
        "df[\"Age\"] = df[\"Age\"].fillna(df[\"Age\"].median())\n",
        "change_log.append([\"Age missing\", \"Filled with median age\", age_nulls])\n",
        "\n",
        "emb_nulls = df[\"Embarked\"].isnull().sum()\n",
        "df[\"Embarked\"] = df[\"Embarked\"].fillna(df[\"Embarked\"].mode()[0])\n",
        "change_log.append([\"Embarked missing\", \"Filled with mode\", emb_nulls])\n",
        "\n",
        "cabin_nulls = df[\"Cabin\"].isnull().sum()\n",
        "df = df.drop(columns=[\"Cabin\"])\n",
        "change_log.append([\"Cabin mostly missing\", \"Dropped the column\", cabin_nulls])\n",
        "\n",
        "# ---------- Step 6: Fix data types ----------\n",
        "df[\"Age\"] = df[\"Age\"].astype(int)\n",
        "df[\"Survived\"] = df[\"Survived\"].astype(\"category\")\n",
        "df[\"Pclass\"] = df[\"Pclass\"].astype(\"category\")\n",
        "df[\"Sex\"] = df[\"Sex\"].astype(\"category\")\n",
        "df[\"Embarked\"] = df[\"Embarked\"].astype(\"category\")\n",
        "change_log.append([\"Wrong data types\", \"Age -> int, Survived/Pclass/Sex/Embarked -> category\", \"5 columns\"])\n",
        "\n",
        "# ---------- Step 7: Outliers in Fare ----------\n",
        "q1 = df[\"Fare\"].quantile(0.25)\n",
        "q3 = df[\"Fare\"].quantile(0.75)\n",
        "iqr = q3 - q1\n",
        "outliers = df[(df[\"Fare\"] < q1 - 1.5 * iqr) | (df[\"Fare\"] > q3 + 1.5 * iqr)]\n",
        "print(\"\\nFare outliers found:\", len(outliers))\n",
        "change_log.append([\"Fare outliers\", \"Checked with IQR, kept them (high fares are real 1st class tickets)\", len(outliers)])\n",
        "\n",
        "# ---------- Step 8: Check again ----------\n",
        "print(\"\\n--- After cleaning ---\")\n",
        "print(\"Nulls left:\", df.isnull().sum().sum())\n",
        "print(\"Duplicates left:\", df.duplicated().sum())\n",
        "df.info()\n",
        "\n",
        "# ---------- Step 9: Save files ----------\n",
        "df.to_csv(\"titanic_cleaned.csv\", index=False)\n",
        "log_df = pd.DataFrame(change_log, columns=[\"Issue\", \"Fix\", \"Count\"])\n",
        "log_df.to_csv(\"change_log.csv\", index=False)\n",
        "print(log_df)\n",
        "print(\"\\nFiles saved: titanic_cleaned.csv and change_log.csv\")"
